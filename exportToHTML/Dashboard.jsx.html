// Simple, transparent match score based on major + overlapping interests/skills
function computeMatchScore(user, job) {
  const majorMatch = user?.major && job.majors?.includes(user.major) ? 40 : 0
  const interests = new Set((user?.interests || []).map((s) => s.toLowerCase()))
  const skills = new Set((job.skills || []).map((s) => s.toLowerCase()))
  let overlap = 0
  for (const i of interests) if (skills.has(i)) overlap++
  const interestScore = Math.min(60, overlap * 10)
  return Math.min(100, majorMatch + interestScore)
}

export default function DashboardPage({ user, onLogout }) {
  // Local state for Saved and Applied
  const [saved, setSaved] = useState([])
  const [applied, setApplied] = useState([])

  // Sort jobs by personalized match score
  const matches = useMemo(() => {
    const withScores = SAMPLE_JOBS.map((j) => ({
      ...j,
      matchScore: computeMatchScore(user, j),
    }))
    return withScores.sort((a, b) => b.matchScore - a.matchScore)
  }, [user])

  const savedIds = useMemo(() => new Set(saved.map((j) => String(j._id))), [saved])
  const appliedIds = useMemo(() => new Set(applied.map((j) => String(j._id))), [applied])

  const handleSave = (job) => {
    if (appliedIds.has(String(job._id)) || savedIds.has(String(job._id))) return
    setSaved((prev) => [job, ...prev])
    // Later: POST /api/jobs/:id/save with userId
  }

  const handleApply = (job) => {
    if (appliedIds.has(String(job._id))) return
    setApplied((prev) => [job, ...prev])
    setSaved((prev) => prev.filter((j) => String(j._id) !== String(job._id)))
    // Later: POST /api/jobs/:id/apply with userId
  }

  return (<>
      {/* FAMU brand top bar */}
      <div className="topbar">
          <div className="brand">
              <span className="brand-badge">FAMU</span>
              <span className="brand-text">Career Assistant</span>
          </div>
          <div className="topbar-user">
              {user?.email || 'Guest'}
          </div>
      </div>

      <div className="dashboard">
      <div className="dashboard__main">
        <header className="dashboard__header">
          <div>
            <h2 className="dashboard__title">Dashboard</h2>
            <p className="dashboard__subtitle">
              Personalized for {user?.major || 'your major'}
              {user?.interests?.length ? ` • Interests: ${user.interests.join(', ')}` : ''}
            </p>
          </div>
          <button className="btn btn--ghost" onClick={() => (typeof onLogout === 'function' ? onLogout() : (window.location.hash = '#/login'))}>
            Back to Login
          </button>
        </header>

        <section className="dashboard__section">
          <h3 className="section__title">Top Matches</h3>
          <div className="job-grid">
            {matches.map((job) => (
              <JobCard
                key={job._id}
                job={job}
                isSaved={savedIds.has(String(job._id))}
                isApplied={appliedIds.has(String(job._id))}
                onSave={handleSave}
                onApply={handleApply}
              />
            ))}
          </div>
        </section>
      </div>

          <aside className="dashboard__sidebar">
              <SavedAppliedPanel saved={saved} applied={applied} />
          </aside>
      </div>
      </>
  )
}
